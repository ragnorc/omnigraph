query employers() {
    match { $p: Person, optional { $c: Company, $p worksAt $c } }
    return { $p.handle, $c }
}

query edges() {
    match { $p: Person, $p $k:(knows | likes) $q }
    return { $p.handle, $k }
}

query fused_per_author($s: String, $v: Vector(3)) {
    match {
        $p: Person,
        let $pop = { match { $d: Doc, $p authored $d } return { $d } },
        from rrf(bm25($pop, $d.body, $s), nearest($pop, $d.embedding, $v, k: 1)) as $f
    }
    return { $p.handle, $d.id, $f }
    order { $p.handle, $f desc }
}
