query nearest_per_author($v: Vector(3)) {
  match {
    $p: Person,
    from nearest({ match { $d: Doc, $p authored $d } return { $d } }, $d.embedding, $v, k: 2) as $s
  }
  return { $p.name, $d.title, $s }
  order { $p.name, $s }
}

query knows_edges() {
  match {
    $c: Person,
    from union({ match { $q: Person, $c $k:knows $q } return { $k } },
               { match { $a: Person, $a $e:knows $c } return { $e as k } })
  }
  return { $c.handle, $k.@src, $k.@dst }
  order { $c.handle, $k }
}

query hits($s: String) {
  match { $d: Doc }
  return { $d.id, match_terms($d.body, terms($s, mode: any)) as hit }
  order { $d.id }
}
