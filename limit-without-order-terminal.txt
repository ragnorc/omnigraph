queries (GQ 3 text; edition 2 writes the filter as a match clause):
query scan_limit() { match { $d: Doc } return { $d.slug, $d.year } limit 10 }
query scan_order_limit() { match { $d: Doc } return { $d.slug, $d.year } order { $d.year desc } limit 10 }
query filter_limit() { match { $d: Doc } where $d.year > 2005 return { $d.slug, $d.year } limit 10 }
query filter_order_limit() { match { $d: Doc } where $d.year > 2005 return { $d.slug, $d.year } order { $d.year desc } limit 10 }
